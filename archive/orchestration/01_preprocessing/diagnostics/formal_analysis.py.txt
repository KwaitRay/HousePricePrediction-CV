"""Frozen train-derived error groups and post-run paired diagnostics (no model fitting)."""
from pathlib import Path
import argparse
import sys
import json
import numpy as np
import pandas as pd
from PIL import Image, ImageOps, ImageDraw

STAGE = Path(__file__).resolve().parents[1]
PROJECT = STAGE.parents[2]
CODE = PROJECT / 'scripts/experiments'
sys.path.insert(0, str(CODE))
from common.config import read_json, write_json, digest, path_at_root
from evaluation.report import compare_runs

def setup():
    output = STAGE / 'diagnostics/frozen_groups'
    output.mkdir(parents=True, exist_ok=False)
    split_path = CODE/'artifacts/split_v1.csv'
    feature_path = PROJECT/'analysis_outputs/20260921T042923Z_7a378d1a/tables/train_image_features.csv'
    split = pd.read_csv(split_path)
    features = pd.read_csv(feature_path)
    frame = split.merge(features[['imageid','width','height','brightness','laplacian_variance']],on='imageid',validate='one_to_one')
    frame['padding'] = 1 - np.rint(np.minimum(frame.width,frame.height)*224/np.maximum(frame.width,frame.height))/224
    training = frame.loc[frame.partition=='train']
    thresholds = {col: training[col].quantile([.2,.8]).tolist() for col in ['price','brightness','laplacian_variance','padding']}
    for col,(lo,hi) in thresholds.items():
        frame[col+'_band'] = np.where(frame[col]<=lo,'low',np.where(frame[col]>=hi,'high','middle'))
    frame['orientation'] = np.where(frame.width>frame.height,'landscape',np.where(frame.width<frame.height,'portrait','square'))
    frame['source_long_edge'] = np.where(np.maximum(frame.width,frame.height)<320,'below_320','at_least_320')
    frame['price_tail'] = np.where(frame.price>1300,'tail','non_tail')
    frame.to_csv(output/'groups.csv',index=False,encoding='utf-8-sig')
    write_json(output/'thresholds.json',{'thresholds':thresholds,'fit_n':len(training),'split_sha256':digest(split_path),
        'feature_table_sha256':digest(feature_path),'quantiles':[.2,.8],'tie_rule':'<=lower is low; >=upper is high; otherwise middle',
        'purpose':'Exploratory error attribution; never replaces complete validation MSE. Thresholds fit exclusively on frozen training side.'})
    print(json.dumps(thresholds,indent=2))

def summarize(run):
    history=pd.read_csv(run/'history.csv')
    cost=read_json(run/'cost.json')
    metrics=read_json(run/'metrics.json')
    train=read_json(run/'train_evaluation/metrics.json')
    checks=read_json(run/'prediction_checks.json')
    return {'run':str(run),'metrics':metrics,'train_metrics_at_best_checkpoint':train,'best_epoch':cost['best_epoch'],
        'last_validation_mse':float(history.val_mse.iloc[-1]),'first_validation_mse':float(history.val_mse.iloc[0]),
        'last_training_data_loss':float(history.data_loss.iloc[-1]),
        'late_validation_mse_mean':float(history.val_mse.tail(10).mean()),
        'late_validation_mse_std':float(history.val_mse.tail(10).std()),
        'generalization_gap_mse':metrics['mse']-train['mse'],'cost':cost,'prediction_checks':checks}

def analyze(run,control=None):
    run=Path(run).resolve()
    cfg=read_json(run/'config.json')
    runtime_cache=None
    if (run/'runtime_cache.json').exists():
        sys.path.insert(0,str(STAGE/'cache_optimization'))
        from cache_evidence import verify_runtime_record
        runtime_cache=verify_runtime_record(run)
    if read_json(run/'status.json')!={'status':'completed','kind':'development'}:
        raise ValueError('Only completed development experiments may be analyzed')
    output=STAGE/'analyses'/cfg['experiment']['id']/run.name
    if control:
        control=Path(control).resolve()
        compare_runs([control,run],output,paired=True)
    else:
        output.mkdir(parents=True,exist_ok=False)
    groups=pd.read_csv(STAGE/'diagnostics/frozen_groups/groups.csv')
    groups=groups.loc[groups.partition=='validation'].drop(columns=['price','group_id','pixel_hash','partition'])
    predictions=pd.read_csv(run/'predictions.csv')
    assert len(predictions)==1601 and not predictions.imageid.duplicated().any()
    assert set(predictions.imageid)==set(groups.imageid)
    recorded=read_json(run/'metrics.json')
    independent_error=predictions.prediction.to_numpy()-predictions.price.to_numpy()
    assert np.isfinite(independent_error).all()
    assert np.isclose(np.mean(independent_error**2),recorded['mse'],rtol=1e-10)
    assert np.isclose(np.mean(abs(independent_error)),recorded['mae'],rtol=1e-10)
    assert np.isclose(np.mean(independent_error[predictions.price.to_numpy()>1300]**2),recorded['tail_mse'],rtol=1e-10)
    frame=predictions.merge(groups,on='imageid',validate='one_to_one')
    summary=summarize(run)
    if runtime_cache:
        summary['runtime_cache']=runtime_cache
    if control:
        baseline=pd.read_csv(control/'predictions.csv')[['imageid','price','squared_error','prediction']].rename(
            columns={'price':'control_price','squared_error':'control_squared_error','prediction':'control_prediction'})
        frame=frame.merge(baseline,on='imageid',validate='one_to_one')
        assert np.array_equal(frame.price.to_numpy(),frame.control_price.to_numpy())
        frame['squared_error_change']=frame.squared_error-frame.control_squared_error
        summary['control']=summarize(control)
        summary['bootstrap']=read_json(output/'paired_bootstrap.json')
        summary['mse_relative_change']=(summary['metrics']['mse']/summary['control']['metrics']['mse']-1)
        init=[read_json(p/'model_summary.json')['initialization_sha256'] for p in [control,run]]
        summary['same_initialization']=init[0]==init[1]
        if not summary['same_initialization']:
            raise ValueError('Preprocessing-only control must have identical initialization')
        influential=frame.squared_error_change.abs().nlargest(10).index
        summary['influence_check']={'top10_absolute_change_sum':float(frame.loc[influential,'squared_error_change'].sum()),
            'total_squared_error_change':float(frame.squared_error_change.sum()),
            'remaining_1591_mean_change':float(frame.drop(index=influential).squared_error_change.mean()),
            'note':'Sensitivity diagnosis only; all 1601 samples remain in the official MSE.'}
        cases=pd.concat([frame.nsmallest(6,'squared_error_change').assign(case_type='improved'),
                         frame.nlargest(6,'squared_error_change').assign(case_type='worsened')])
    else:
        cases=frame.nlargest(12,'squared_error').assign(case_type='worst_baseline')
    columns=['price_band','price_tail','brightness_band','laplacian_variance_band','padding_band','orientation','source_long_edge']
    grouped=[]
    for column in columns:
        for name,part in frame.groupby(column):
            row={'grouping':column,'group':name,'n':len(part),'mse':float(part.squared_error.mean()),
                 'mae':float(part.error.abs().mean()),'signed_bias':float(part.error.mean()),
                 'mean_true_price':float(part.price.mean()),'tail_fraction':float((part.price>1300).mean()),
                 'sse_share':float(part.squared_error.sum()/frame.squared_error.sum())}
            if control:
                row.update(control_mse=float(part.control_squared_error.mean()),
                           mse_change=float(part.squared_error_change.mean()),
                           sse_change=float(part.squared_error_change.sum()))
            grouped.append(row)
    pd.DataFrame(grouped).to_csv(output/'group_metrics.csv',index=False,encoding='utf-8-sig')
    frame.to_csv(output/'diagnostic_predictions.csv',index=False,encoding='utf-8-sig')
    cases.to_csv(output/'selected_cases.csv',index=False,encoding='utf-8-sig')
    sheet=Image.new('RGB',(4*280,3*250),'white')
    draw=ImageDraw.Draw(sheet)
    root=path_at_root(cfg['data']['root'])
    for i,(_,row) in enumerate(cases.iterrows()):
        with Image.open(root/'train'/row.imageid) as image:
            image=ImageOps.contain(ImageOps.exif_transpose(image).convert('RGB'),(276,190))
            x,y=(i%4)*280,(i//4)*250
            sheet.paste(image,(x,y))
            draw.text((x,y+192),f'{row.imageid} {row.case_type}',fill='black')
            draw.text((x,y+207),f'true={row.price:.1f} pred={row.prediction:.1f}',fill='black')
            if control:
                draw.text((x,y+222),f'control={row.control_prediction:.1f} dSE={row.squared_error_change:.0f}',fill='black')
    sheet.save(output/'cases.png')
    import matplotlib
    matplotlib.use('Agg')
    import matplotlib.pyplot as plt
    plotted_runs = [(control,'Control'),(run,'Candidate')] if control else [(run,'AlexNet baseline')]
    panels=[]
    for source,label in plotted_runs:
        panels.extend([(pd.read_csv(source/'train_evaluation/predictions.csv'),label+' / training'),
                       (pd.read_csv(source/'predictions.csv'),label+' / validation')])
    upper=250*np.ceil(max(max(p.price.max(),p.prediction.max()) for p,_ in panels)/250)
    fig,axes=plt.subplots(len(plotted_runs),2,figsize=(10,4.5*len(plotted_runs)),squeeze=False)
    for ax,(p,label) in zip(axes.flat,panels):
        for tail,color in [(False,'tab:blue'),(True,'tab:orange')]:
            part=p.loc[(p.price>1300)==tail]
            ax.scatter(part.price,part.prediction,s=6,alpha=.3,color=color,
                       label=f"{'Tail' if tail else 'Non-tail'} n={len(part)}")
        ax.plot([0,upper],[0,upper],color='black',linestyle='--',linewidth=1)
        ax.set(title=label,xlabel='True price (thousand USD)',ylabel='Predicted price (thousand USD)',
               xlim=(0,upper),ylim=(0,upper))
        ax.legend(fontsize=8)
        ax.grid(alpha=.15)
    fig.suptitle('Best validation checkpoint; identical axes across panels')
    fig.tight_layout()
    fig.savefig(output/'prediction_scatter.png',dpi=140)
    plt.close(fig)
    fig,axes=plt.subplots(1,3,figsize=(15,4))
    for source,label in plotted_runs:
        history=pd.read_csv(source/'history.csv')
        epochs=np.arange(1,len(history)+1)
        axes[0].plot(epochs,history.data_loss,label=label)
        axes[1].plot(epochs,history.val_mse,label=label)
        best=read_json(source/'cost.json')['best_epoch']
        axes[1].scatter([best],[history.val_mse.iloc[best-1]],s=35)
        axes[2].plot(epochs,history.val_tail_mse,label=label)
    for ax,title,ylabel in zip(axes,
            ['Training (dropout active, changing weights)','Validation; dot = selected epoch','Validation tail (>1300)'],
            ['Scaled-target training MSE','Original-price MSE','Original-price tail MSE']):
        ax.set(title=title,xlabel='Epoch',ylabel=ylabel)
        ax.grid(alpha=.15)
        ax.legend(fontsize=8)
    fig.tight_layout()
    fig.savefig(output/'learning_curves.png',dpi=140)
    plt.close(fig)
    write_json(output/'analysis.json',summary)
    write_json(output/'analysis_provenance.json',{'script_sha256':digest(__file__),'runtime_cache_sha256':digest(run/'runtime_cache.json') if runtime_cache else None,'frozen_groups_sha256':digest(STAGE/'diagnostics/frozen_groups/groups.csv'),
             'prediction_sha256':digest(run/'predictions.csv'),'control_prediction_sha256':digest(control/'predictions.csv') if control else None})
    journal='group_reviews_v2.md' if cfg['training'].get('loader_mode')=='persistent' else 'group_reviews.md'
    text=f"# {cfg['experiment']['name']} — 结果诊断\n\n本页为数值诊断，原因判断和下一轮决策见阶段{journal}。\n\n"
    text+=f"验证MSE：{summary['metrics']['mse']:.6f}；MAE：{summary['metrics']['mae']:.6f}；尾部MSE：{summary['metrics']['tail_mse']:.6f}。\n\n"
    text+=f"最佳轮次：{summary['best_epoch']}；同一权重完整训练MSE：{summary['train_metrics_at_best_checkpoint']['mse']:.6f}；训练/验证差：{summary['generalization_gap_mse']:.6f}。\n\n"
    if control:
        text+=f"候选相对对照MSE变化：{summary['mse_relative_change']:.2%}；候选减对照的组重采样95%区间：{summary['bootstrap']['interval_95']}。\n\n"
    text+='[完整分析](analysis.json) · [分组指标](group_metrics.csv) · [逐图诊断](diagnostic_predictions.csv) · [所选正反例](selected_cases.csv)\n\n![学习曲线](learning_curves.png)\n\n![预测与真实价](prediction_scatter.png)\n\n![误差案例](cases.png)\n'
    (output/'README.md').write_text(text,encoding='utf-8')
    print(output)
    print(json.dumps(summary,indent=2,ensure_ascii=False))

if __name__=='__main__':
    parser=argparse.ArgumentParser()
    sub=parser.add_subparsers(dest='command',required=True)
    sub.add_parser('setup')
    p=sub.add_parser('analyze')
    p.add_argument('run')
    p.add_argument('--control')
    args=parser.parse_args()
    if args.command=='setup':setup()
    else:analyze(args.run,args.control)
