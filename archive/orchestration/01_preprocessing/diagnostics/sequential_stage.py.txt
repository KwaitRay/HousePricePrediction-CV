"""Sequential, resumable preprocessing experiments with a documented review gate.

Lives outside frozen training sources. Never changes training settings or retries
failed training silently. Numeric reviews distinguish evidence from hypotheses.
"""

import datetime as dt
import json
import os

import sys
import time

import pandas as pd

STAGE = Path(__file__).resolve().parents[1]
RECORD = STAGE.parent
CODE = STAGE.parents[2] / 'scripts/experiments'
sys.path.insert(0, str(CODE))
from common.config import read_json, digest, differences, resolve, validate

BASE = RECORD/'02_baselines/runs/alexnet/20261001T092947Z_71a2f7b6'

CANDIDATES = [
    ('preprocess_01_stretch','preprocess.geometry','去除补边、提高画面占用，同时引入非等比形变；二者作用不能单独分离。'),

STATE = STAGE/'pipeline_state.json'

def now():
    return dt.datetime.now(dt.timezone.utc).isoformat()

def write(path, value):
    path=Path(path)
    path.parent.mkdir(parents=True,exist_ok=True)
    temporary=path.with_name(path.name+'.tmp')
    temporary.write_text(json.dumps(value,indent=2,ensure_ascii=False)+'\n',encoding='utf-8')
    os.replace(temporary,path)

def relative(path):
    return Path(os.path.relpath(path,STAGE)).as_posix()

def config_for(identifier):
    return CODE/('configs/01_baselines/alexnet.json' if identifier=='alexnet' else f'configs/00_preparation/{identifier}.json')

def check_sources():
    env=read_json(BASE/'environment.json')
    for name,expected in env['source_hashes'].items():
        if digest(CODE/name)!=expected:
            raise RuntimeError(f'Frozen training source changed: {name}; stop for review')

def check_plan():
    check_sources()
    baseline=read_json(BASE/'config.json')
    for identifier,field,_ in CANDIDATES:
        cfg=validate(resolve(config_for(identifier)))
        cfg['seed']=2026
        cfg['run_kind']='development'
        diff=differences(baseline,cfg)
        if set(diff)!={field}:
            raise RuntimeError(f'Unexpected controlled difference: {identifier}: {diff}')
    print('All six planned candidates have exactly one registered difference.',flush=True)

def complete_checks(run,control=None):
    if read_json(run/'status.json')!={'status':'completed','kind':'development'}:
        raise RuntimeError(f'Incomplete run: {run}')
    cost=read_json(run/'cost.json')
    assert cost['epochs']==60 and not cost['stopped_early']
    assert cost['optimizer_updates']==12000 and cost['samples_seen']==383940
    history=pd.read_csv(run/'history.csv')
    assert list(history.epoch)==list(range(1,61))
    assert (history.samples==6399).all() and (history.updates==200).all()
    assert np.isfinite(history[['data_loss','val_mse','val_mae','val_tail_mse']].to_numpy()).all()
    for folder,n in [(run,1601),(run/'train_evaluation',6399)]:
        pred=pd.read_csv(folder/'predictions.csv')
        assert len(pred)==n and pred.imageid.is_unique
        assert np.isfinite(pred[['price','prediction']].to_numpy()).all()
        metrics=read_json(folder/'metrics.json')
        assert np.isclose(np.mean((pred.prediction-pred.price)**2),metrics['mse'],rtol=1e-10)
    env=read_json(run/'environment.json')
    ref=read_json(BASE/'environment.json')
    for key in ['source_hashes','packages','gpu','cuda','cudnn']:
        assert env[key]==ref[key],key
    if control:
        assert read_json(run/'model_summary.json')['initialization_sha256']==read_json(control/'model_summary.json')['initialization_sha256']
        a,b=read_json(control/'data_manifest.json'),read_json(run/'data_manifest.json')
        assert a['split_sha256']==b['split_sha256']

def analyze(run,control=None):
    cfg=read_json(run/'config.json')
    output=STAGE/'analyses'/cfg['experiment']['id']/run.name
    if not (output/'analysis.json').exists():
        if output.exists():
            raise RuntimeError(f'Partial analysis requires review, not overwrite: {output}')
        log=STAGE/'logs'/f'analysis_{run.name}.log'
        command=[sys.executable,'-X','utf8',str(STAGE/'diagnostics/formal_analysis.py'),'analyze',str(run)]
        if control:
            command+=['--control',str(control)]
        with log.open('x',encoding='utf-8') as stream:
            subprocess.run(command,stdout=stream,stderr=subprocess.STDOUT,check=True,cwd=CODE)
    provenance=read_json(output/'analysis_provenance.json')
    assert provenance['prediction_sha256']==digest(run/'predictions.csv')
    if control:
        assert provenance['control_prediction_sha256']==digest(control/'predictions.csv')
    return output

def wait_run(run,child=None):
    last=0
    while True:
        try:
            status=read_json(run/'status.json')
        except (FileNotFoundError,json.JSONDecodeError):
            status={}
        if status.get('status')=='completed':
            if child and child.wait()!=0:
                raise RuntimeError('Training process failed after completed status')
            return
        if status.get('status')=='failed':
            raise RuntimeError(f'Training failed; inspect {run}')
        if child and child.poll() is not None:
            raise RuntimeError(f'Training exited without completed status: {child.returncode}')
        if time.monotonic()-last>60:
            last=time.monotonic()
            history=run/'history.csv'
            try:
                h=pd.read_csv(history)
                progress={'completed_epochs':int(h.epoch.iloc[-1]),'best_validation_mse_so_far':float(h.val_mse.min())}
            except (FileNotFoundError,pd.errors.EmptyDataError,IndexError):
                progress={'completed_epochs':0}
            write(STAGE/'execution_state.json',{'status':'training','active_run':relative(run),'snapshot_utc':now(),
                'review_before_next':True,'protocol':'protocol.md','group_reviews':'group_reviews.md',**progress})
            newest=max(p.stat().st_mtime for p in run.iterdir() if p.is_file())
            if time.time()-newest>1200:
                raise RuntimeError('No run artifact progress for 20 minutes; inspect instead of starting another GPU job')
        time.sleep(10)

def get_run(state,identifier,seed):
    key=f'{identifier}:{seed}'
    if key in state['runs']:
        run=RECORD/state['runs'][key]
        wait_run(run)
        complete_checks(run)
        return run
    check_sources()
    root=(RECORD/'02_baselines' if identifier=='alexnet' else STAGE)/'runs'/identifier
    before=set(root.glob('*')) if root.exists() else set()
    # Any untracked running experiment is reviewed manually before another launch.
    for p in before:
        if read_json(p/'status.json').get('status')=='running':
            raise RuntimeError(f'Untracked active run requires review: {p}')
    logstage=RECORD/'02_baselines' if identifier=='alexnet' else STAGE
    log=logstage/'logs'/f'{identifier}_seed{seed}_pipeline.log'
    with log.open('x',encoding='utf-8') as stream:
        child=subprocess.Popen([sys.executable,'-X','utf8','run.py','train',str(config_for(identifier)),'--seed',str(seed)],
            cwd=CODE,stdout=stream,stderr=subprocess.STDOUT)
        for _ in range(60):
            added=set(root.glob('*'))-before if root.exists() else set()
            if len(added)==1:
                run=added.pop()
                state['runs'][key]=os.path.relpath(run,RECORD).replace('\\','/')
                state['active']=key
                write(STATE,state)
                break
            if len(added)>1 or child.poll() is not None:
                raise RuntimeError(f'Unexpected launch result; inspect {log}')
            time.sleep(1)
        else:
            raise RuntimeError('Training did not create a run directory; inspect child before resuming')
        with (STAGE/'group_reviews.md').open('a',encoding='utf-8') as journal:
            journal.write(f'\n### 已按顺序启动 {identifier}，seed {seed}\n\n前一组数值review已落盘。[运行]({relative(run)})；[日志]({relative(log)})。\n')
        print(f'Started {key}: {run.name}',flush=True)
        wait_run(run,child)
    complete_checks(run)
    return run

def finish(state,challengers):
    rows=[]
    for identifier in ['alexnet',*challengers]:
        for seed in [2026,2027,2028]:
            run=RECORD/state['runs'][f'{identifier}:{seed}']
            complete_checks(run)
            rows.append({'method':identifier,'seed':seed,**read_json(run/'metrics.json'),**read_json(run/'cost.json')})
    frame=pd.DataFrame(rows)
    output=STAGE/'final_comparison'
    output.mkdir(exist_ok=True)
    frame.to_csv(output/'seed_results.csv',index=False,encoding='utf-8-sig')
    summary=frame.groupby('method')[['mse','mae','tail_mse','train_seconds','peak_cuda_bytes']].agg(['mean','std'])
    summary.to_csv(output/'summary.csv',encoding='utf-8-sig')
    base=frame.loc[frame.method=='alexnet'].set_index('seed')
    eligibility=[]
    for name in challengers:
        candidate=frame.loc[frame.method==name].set_index('seed')
        changes=candidate.mse-base.mse
        eligibility.append({'method':name,'mean_mse':float(candidate.mse.mean()),'mean_paired_change':float(changes.mean()),
            'seeds_improved':int((changes<0).sum()),'per_seed_changes':changes.to_dict(),
            'eligible':bool(changes.mean()<0 and (changes<0).sum()>=2)})
    eligible=[r for r in eligibility if r['eligible']]
    winner=min(eligible,key=lambda r:r['mean_mse'])['method'] if eligible else 'alexnet'
    decision={'selected_method':winner,'challengers':challengers,'eligibility':eligibility,
        'rule':'Lower mean MSE and at least 2 of 3 seed improvements; otherwise retain baseline.',
        'limitations':'Three seeds on one repeatedly used development split; no independent test claim; close ranks remain uncertain.',
        'completed_utc':now()}
    write(output/'decision.json',decision)
    screening=read_json(STAGE/'screening_selection.json')
    lines=['# 预处理备选阶段：三种子比较与选择','',f'按预登记规则保留 **{winner}**。',
        '先完成六项seed2026初筛，再对初筛最佳候选及距离其1%以内的候选，与原基线一起复核2027/2028。每次运行均在前组数值review落盘后启动。','',
        '## 六项初筛（seed2026）','',
        '| 方法 | 验证MSE | 相对原基线变化 |', '| --- | ---: | ---: |']
    baseline_score=read_json(BASE/'metrics.json')['mse']
    for name,score in sorted(screening['scores'].items(),key=lambda item:item[1]):
        lines.append(f'| {name} | {score:.2f} | {score/baseline_score-1:+.2%} |')
    lines += ['', '## 三种子复核','',
        '| 方法 | MSE均值 ± 标准差 | MAE均值 | 尾部MSE均值 | 平均分钟 |', '| --- | ---: | ---: | ---: | ---: |']
    for name in ['alexnet',*challengers]:
        x=frame.loc[frame.method==name]
        lines.append(f'| {name} | {x.mse.mean():.2f} ± {x.mse.std():.2f} | {x.mae.mean():.2f} | {x.tail_mse.mean():.2f} | {x.train_seconds.mean()/60:.2f} |')
    lines+=['','## 逐种子一致性','']
    for row in eligibility:
        lines.append(f"- {row['method']}：候选减对照MSE {row['per_seed_changes']}；{row['seeds_improved']}/3个种子改善，平均变化{row['mean_paired_change']:+.2f}。")
    lines+=['','若差距很小或种子间方向变化，保留规则只是工程选择，不证明存在稳定优势。均值/标准差来自单次运行指标，未对预测做集成。未复核的候选只有一个种子的结果，不能宣称已证明其总体劣于入围者。',
        '', '[逐种子完整指标](seed_results.csv) · [均值及样本标准差](summary.csv) · [选择规则结果](decision.json) · [逐组原因分析](../group_reviews.md)',
        '', '结论范围：当前AlexNet、当前固定划分及60轮预算。开发验证集已用于多次选择，不等于独立测试泛化结论。']
    (output/'README.md').write_text('\n'.join(lines)+'\n',encoding='utf-8')
    chosen=RECORD/state['runs'][f'{winner}:2026']
    target=CODE/'configs/selected/input.json'
    reason=f'Preprocessing three-seed rule; evidence: {output / "README.md"}; selected={winner}; no test data used.'
    if target.exists():
        assert Path(read_json(target)['selection']['run']).resolve()==chosen.resolve(), 'Existing selected input differs; do not overwrite'
    else:
        subprocess.run([sys.executable,'-X','utf8','run.py','select',str(chosen),'--slot','input','--reason',reason],cwd=CODE,check=True)
        selected=read_json(target)
        selected['selection']['warning']='按预登记规则完成三个训练种子的开发集复核；仍非独立测试结论。'
        selected['selection']['three_seed_evidence']=str(output/'decision.json')
        write(target,selected)
    for doc in [STAGE/'README.md',RECORD/'README.md']:
        link=os.path.relpath(output/'README.md',doc.parent).replace('\\','/')
        original=doc.read_text(encoding='utf-8')
        title,rest=original.split('\n',1)
        doc.write_text(f'{title}\n\n**正式预处理比较已完成；按三种子规则保留 {winner}。** [最终比较与限制]({link})。下文保留阶段历史。\n'+rest,encoding='utf-8')
    state.update(status='completed',selected_method=winner,completed_utc=now())
    write(STATE,state)
    write(STAGE/'execution_state.json',{'status':'completed','selected_method':winner,'final_report':'final_comparison/README.md','snapshot_utc':now()})

def main():
    parser=argparse.ArgumentParser()
    parser.add_argument('command',choices=['check','run'])
    args=parser.parse_args()
    check_plan()
    if args.command=='check':
        return
    import msvcrt
    lock=(STAGE/'pipeline.lock').open('a+b')
    lock.seek(0)
    if not lock.read(1):
        lock.write(b'0');lock.flush()
    lock.seek(0)
    msvcrt.locking(lock.fileno(),msvcrt.LK_NBLCK,1)
    state=read_json(STATE) if STATE.exists() else {'status':'running','started_utc':now(),'runs':{
        'alexnet:2026':os.path.relpath(BASE,RECORD).replace('\\','/'),
        'preprocess_01_stretch:2026':os.path.relpath(ACTIVE,RECORD).replace('\\','/')}}
    if state['status']=='completed':
        return
    write(STATE,state)
    try:
        complete_checks(BASE)
        # The initial baseline has already received manual review before stretch.
        assert '### 下一组决策（启动前登记）' in (STAGE/'group_reviews.md').read_text(encoding='utf-8')
        for identifier,_,mechanism in CANDIDATES:
            run=get_run(state,identifier,2026)
            review(run,BASE,mechanism)
        scores={identifier:read_json(RECORD/state['runs'][f'{identifier}:2026']/'metrics.json')['mse'] for identifier,_,_ in CANDIDATES}
        best=min(scores.values())
        challengers=[name for name,score in scores.items() if score<=best*1.01]
        write(STAGE/'screening_selection.json',{'scores':scores,'challengers':challengers,'threshold':best*1.01,'selected_utc':now()})
        for seed in [2027,2028]:
            baseline=get_run(state,'alexnet',seed)
            review(baseline,None,'')
            for identifier in challengers:
                mechanism=next(m for name,_,m in CANDIDATES if name==identifier)
                candidate=get_run(state,identifier,seed)
                review(candidate,baseline,mechanism)
        finish(state,challengers)
    except Exception:
        state.update(status='stopped_for_review',error=traceback.format_exc(),stopped_utc=now())
        write(STATE,state)
        write(STAGE/'execution_state.json',{'status':'stopped_for_review','pipeline_state':'pipeline_state.json','snapshot_utc':now(),'error':state['error']})
        raise
    finally:
        lock.close()

if __name__=='__main__':
    main()
