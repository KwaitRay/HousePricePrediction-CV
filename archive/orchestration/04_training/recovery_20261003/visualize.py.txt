"""Read-only metrics plotting; never imports or modifies the training process."""
from pathlib import Path
import json,os,sys,time,hashlib
import pandas as pd
import matplotlib
matplotlib.use('Agg')
from matplotlib import pyplot as plt
from matplotlib import font_manager

STAGE=Path(__file__).resolve().parent.parent
font_path=Path('C:/Windows/Fonts/msyh.ttc')
if font_path.exists():
    font_manager.fontManager.addfont(str(font_path))
    plt.rcParams['font.family']=font_manager.FontProperties(fname=str(font_path)).get_name()
plt.rcParams['axes.unicode_minus']=False

def plot(run):
    run=Path(run);history=run/'history.csv'
    if not history.exists():return
    raw=history.read_bytes()
    try:frame=pd.read_csv(history)
    except (pd.errors.EmptyDataError,pd.errors.ParserError):return
    if frame.empty or frame[['epoch','data_loss','val_mse','val_mae','val_tail_mse']].isna().any().any():return
    # A live writer can be between CSV writes; retry on the next watcher pass.
    if history.read_bytes()!=raw:return
    best=frame.loc[frame.val_mse.idxmin()];x=frame.epoch
    fig,axes=plt.subplots(2,3,figsize=(16,8))
    panels=[('训练损失（价格已缩放）','data_loss'),('验证均方误差（原价格尺度）','val_mse'),
            ('验证平均绝对误差','val_mae'),('高价尾部均方误差','val_tail_mse'),
            ('固定32张训练图诊断误差','probe_mse'),('每轮耗时（秒）','seconds')]
    for ax,(title,col) in zip(axes.flat,panels):
        ax.plot(x,frame[col]);ax.axvline(best.epoch,color='tab:red',linestyle='--',alpha=.7)
        ax.set(title=title,xlabel='训练轮次');ax.grid(alpha=.2)
    fig.suptitle(f'验证集当前最佳：第{int(best.epoch)}轮，均方误差{best.val_mse:.2f}；已完成{len(frame)}轮\n红虚线为验证集最佳轮次；训练诊断曲线不用于选取权重')
    fig.tight_layout(rect=(0,0,1,.92))
    dest=run/'指标总览.png';temp=run/'指标总览.tmp.png';fig.savefig(temp,dpi=130);plt.close(fig);os.replace(temp,dest)
    # Full training evaluation only exists after restoring the best validation checkpoint.
    train=run/'train_evaluation/metrics.json';val=run/'metrics.json'
    if train.exists() and val.exists():
        a=json.loads(train.read_text(encoding='utf-8'));b=json.loads(val.read_text(encoding='utf-8'))
        fig,axes=plt.subplots(1,3,figsize=(12,4))
        for ax,key,label in zip(axes,['mse','mae','tail_mse'],['均方误差','平均绝对误差','高价尾部均方误差']):
            values=[a[key],b[key]];ax.bar(['完整训练集','完整验证集'],values,color=['tab:blue','tab:orange']);ax.set_title(label)
            for i,v in enumerate(values):ax.text(i,v,f'{v:.1f}',ha='center',va='bottom',fontsize=9)
            ax.set_ylim(0,max(values)*1.15)
        fig.suptitle('同一验证集最佳权重：完整训练与验证误差');fig.tight_layout(rect=(0,0,1,.9))
        temp=run/'泛化差距.tmp.png';fig.savefig(temp,dpi=130);plt.close(fig);os.replace(temp,run/'泛化差距.png')
    (run/'visualization_provenance.json').write_text(json.dumps({'source':'history.csv','history_sha256':hashlib.sha256(raw).hexdigest(),
        'epochs':len(frame),'best_validation_epoch':int(best.epoch),'script_sha256':hashlib.sha256(Path(__file__).read_bytes()).hexdigest()},indent=2),encoding='utf-8')

def main():
    seen={}
    while True:
        for h in (STAGE/'runs').glob('*/*/history.csv'):
            try:
                stamp=h.stat().st_mtime_ns
                metric=h.parent/'metrics.json';signature=(stamp,metric.stat().st_mtime_ns if metric.exists() else None)
                if seen.get(str(h))!=signature:plot(h.parent);seen[str(h)]=signature
            except (OSError,ValueError,KeyError,pd.errors.ParserError) as error:
                print(h,error,flush=True)
        if '--watch' not in sys.argv:break
        state=STAGE/'execution_state.json'
        if state.exists():
            try:
                if json.loads(state.read_text(encoding='utf-8')).get('status') in {'extended_optimizer_block_completed','failed'}:break
            except (ValueError,OSError):pass
        time.sleep(60)

if __name__=='__main__':main()
